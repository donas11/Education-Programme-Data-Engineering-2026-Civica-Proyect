Modelos Incrementales & Snapshots SCD2

⚡ Modelos Incrementales
test_benchmark
→ SILVER.TEST_BENCHMARK
unique_key = 'id_test'
WHERE fecha > MAX(fecha)
-- Solo scores nuevos
precio_modelo
→ SILVER.PRECIO_MODELO
unique_key = 'id_precio_modelo'
WHERE fecha > MAX(fecha)
-- Solo precios nuevos
📸 Snapshots SCD2
snp_modelo
→ SNAPSHOT.SNP_MODELO
strategy = 'check'
unique_key = 'id_modelo'
check_cols: nombre_comercial,
 context_window, multimodal,
 opensource, fecha_lanzamiento
snp_precio_modelo
→ SNAPSHOT.SNP_PRECIO_MODELO
strategy = 'check'
unique_key = 'id_precio_modelo'
check_cols: precio_M_entrada,
 precio_M_salida, tipo_plan
Ambos modelos usan on_schema_change = 'append_new_columns' — compatibilidad futura garantizada
